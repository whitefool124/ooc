using System;
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.UI;

namespace OCC.Combat.Presentation
{
    // Pinned paper reacts to intent to open a new archive; it never represents a saved result.
    [RequireComponent(typeof(CanvasRenderer))]
    public sealed class ArchivePaperDetail : MaskableGraphic
    {
        private Func<float> intensity;
        private bool pulling;
        private float progress;
        private int drawnStep = -1;
        public float Progress => progress;

        public static ArchivePaperDetail Create(Transform parent, Func<float> motion)
        {
            var go = new GameObject("夹住的空白档案纸", typeof(RectTransform), typeof(CanvasRenderer));
            go.transform.SetParent(parent, false);
            var rect = (RectTransform)go.transform;
            rect.anchorMin = rect.anchorMax = new Vector2(0f, 1f);
            rect.anchoredPosition = new Vector2(916f, -598f);
            rect.sizeDelta = new Vector2(224f, 136f);
            var detail = go.AddComponent<ArchivePaperDetail>();
            detail.intensity = motion;
            detail.raycastTarget = false;
            return detail;
        }

        public void SetPulling(bool value) => pulling = value;

        private void Update() => Advance(Time.unscaledDeltaTime);

        public void Advance(float seconds)
        {
            float motion = intensity == null ? 0f : intensity();
            progress = motion <= 0f ? 0f : Mathf.MoveTowards(progress, pulling ? 1f : 0f,
                Mathf.Max(0f, seconds) / ((pulling ? .28f : .42f) * Mathf.Max(.35f, motion)));
            int step = Mathf.RoundToInt(progress * 8f);
            if (step == drawnStep) return;
            drawnStep = step;
            SetVerticesDirty();
        }

        protected override void OnDisable()
        {
            pulling = false;
            progress = 0f;
            drawnStep = -1;
            base.OnDisable();
        }

        protected override void OnPopulateMesh(VertexHelper vh)
        {
            vh.Clear();
            float t = Mathf.Round(progress * 8f) / 8f;
            float dx = Mathf.Round(t * 12f / 2f) * 2f;
            float curl = Mathf.Round(t * 16f / 2f) * 2f;
            Color edge = ArchiveUiStyle.QuietInk;
            Color paper = ArchiveUiStyle.LightPaper;
            Rect(vh, -80, -48, 158, 86, edge);
            Rect(vh, -78, -46, 154, 82, ArchiveUiStyle.Paper);
            Rect(vh, -84, -44, 160, 86, edge);
            Rect(vh, -82, -42, 156, 82, paper);
            float right = 76 - dx;
            // The upper left remains pinned. Only the free lower edge slips and curls.
            Sheet(vh, -86, 44, right, -38, -86 - dx, curl, edge);
            Sheet(vh, -84, 42, right - 2, -36, -84 - dx, Mathf.Max(0, curl - 2), paper);
            if (curl > 0)
            {
                Triangle(vh, new Vector2(right - curl, -38), new Vector2(right, -38 + curl),
                    new Vector2(right - curl, -38 + curl), edge);
                Triangle(vh, new Vector2(right - curl + 2, -34), new Vector2(right - 2, -38 + curl),
                    new Vector2(right - curl + 2, -38 + curl - 2), ArchiveUiStyle.Paper);
            }
            // The clamp stays in precisely the same position through the whole motion.
            Color brass = ArchiveUiStyle.Brass;
            Rect(vh, -64, 26, 18, 28, edge);
            Rect(vh, -62, 28, 14, 24, brass);
            Rect(vh, -58, 32, 6, 16, edge);
            Rect(vh, -56, 34, 2, 12, paper);
            Rect(vh, -68, 26, 26, 4, brass);
        }

        private static void Sheet(VertexHelper vh, float left, float top, float right, float bottom,
            float freeLeft, float curl, Color color)
        {
            Vector2 pin = new Vector2(left, top);
            Triangle(vh, pin, new Vector2(right, top), new Vector2(right, bottom + curl), color);
            Triangle(vh, pin, new Vector2(right, bottom + curl), new Vector2(right - curl, bottom), color);
            Triangle(vh, pin, new Vector2(right - curl, bottom), new Vector2(freeLeft, bottom), color);
        }
        private static void Rect(VertexHelper vh, float x, float y, float w, float h, Color color)
        {
            Triangle(vh, new Vector2(x, y), new Vector2(x, y + h), new Vector2(x + w, y + h), color);
            Triangle(vh, new Vector2(x, y), new Vector2(x + w, y + h), new Vector2(x + w, y), color);
        }
        private static void Triangle(VertexHelper vh, Vector2 a, Vector2 b, Vector2 c, Color color)
        {
            int index = vh.currentVertCount;
            vh.AddVert(a, color, Vector2.zero); vh.AddVert(b, color, Vector2.zero); vh.AddVert(c, color, Vector2.zero);
            vh.AddTriangle(index, index + 1, index + 2);
        }
    }

    public sealed class ArchivePaperPullTrigger : MonoBehaviour, IPointerEnterHandler, IPointerExitHandler
    {
        private ArchivePaperDetail paper;
        public void Configure(ArchivePaperDetail target) => paper = target;
        public void OnPointerEnter(PointerEventData data) { if (paper != null) paper.SetPulling(true); }
        public void OnPointerExit(PointerEventData data) { if (paper != null) paper.SetPulling(false); }
        private void OnDisable() { if (paper != null) paper.SetPulling(false); }
    }
}
